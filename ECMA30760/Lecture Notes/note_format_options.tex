\newcommand{\BodyLead}{%
  \section*{1.1 Introduction}
  Formally, $\succ_i$ is a \emph{binary relation}; i.e., it is a subset of $A \times A$
  with the infix notation of $x \succ_i y$ instead of $(x,y) \in \succ_i$.
}

\newcommand{\SampleDefText}{%
  A social choice function is a mapping $f: P^N \to A$.
}

\newcommand{\SampleExampleText}{%
  For instance, if $A=\{w,x,y,z\}$, the plurality rule chooses $w$, instant runoff
  chooses $x$, elimination chooses $y$, the Borda count chooses $x$, and the Copeland
  rule chooses $x$.
}

\newcommand{\SampleLemmaText}{%
  If $X$ and $Y$ are dominating sets, then $X \subseteq Y$ or $Y \subseteq X$.
}

\newcommand{\SampleTheoremText}{%
  If $|A|>2$, then every surjective and strategy-proof social choice function is
  dictatorial, in that there is some agent $i$ such that
  \[
    f(\succ)=\arg\max_{\succ_i} A
  \]
  for all preference profiles $\succ$.
}

\newcommand{\SampleMarginNote}{%
  \marginpar{%

    \end{tcolorbox}%

\newcommand{\SampleProof}{%
  \noindent\textit{Proof of Lemma 1.1.9.}
  Suppose not. Then any $x\in X\setminus Y$ and $y\in Y\setminus X$ have to both
  beat each other in head-to-head elections, which cannot happen under strict preferences.
  \hfill$\square$\par
}
