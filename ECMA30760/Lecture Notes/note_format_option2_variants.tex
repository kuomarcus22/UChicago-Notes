\newcommand{\DefText}{A social choice function is a mapping $f:P^N\to A$.}
\newcommand{\ExampleText}{For instance, if $A=\{w,x,y,z\}$, plurality chooses $w$,
instant runoff chooses $x$, elimination chooses $y$, the Borda count chooses $x$,
and the Copeland rule chooses $x$.}
\newcommand{\LemmaText}{If $X$ and $Y$ are dominating sets, then $X\subseteq Y$ or $Y\subseteq X$.}
\newcommand{\TheoremText}{If $|A|>2$, then every surjective and strategy-proof social
choice function is dictatorial, in that there is some agent $i$ such that
\[
  f(\succ)=\arg\max_{\succ_i}A
\]
for all preference profiles $\succ$.}
\newcommand{\ProofText}{%
  \noindent\textit{Proof of Lemma 1.1.9.}
  Suppose not. Then any $x\in X\setminus Y$ and $y\in Y\setminus X$ have to both
  beat each other in head-to-head elections, which cannot happen under strict preferences.
  \hfill$\square$\par
}

\VariantTitle{1}{Micro Header Bands}{The closest sibling to Option 2: full-width colored headers, smaller vertical padding, and very faint body tints.}
\DefaultSample

\StyleTwo
\VariantTitle{2}{Band Plus Left Spine}{A full header band keeps the Option 2 scan pattern, while a thin colored spine lets boxes remain recognizable even when the header is far above a multi-page block.}
\DefaultSample

\StyleThree
\VariantTitle{3}{White Cards With Saturated Headers}{The body is fully white and the environment color is concentrated in the header. This is cleaner for long formal statements and proofs.}
\DefaultSample

\StyleFour
\VariantTitle{4}{Overline Bands}{A stronger colored rule sits above each header. This is more editorial: the color reads like a section marker rather than a filled note.}
\DefaultSample

\StyleFive
\VariantTitle{5}{Open Full-Width Ribbons}{The header is a full-width ribbon, but the body is not framed. This is the most page-efficient banded design while still preserving color-coded environment starts.}
\OpenHeadSample

\StyleSix
\VariantTitle{6}{Run-In Tags With Side Rules}{The label begins the paragraph, so short definitions save an entire title line. The side rule preserves the color code when the text wraps.}
\RunInSample

\StyleSeven
\VariantTitle{7}{Inline Header Row}{The title is attached to the upper text row rather than separated by a title/body boundary. This makes compact environments feel closer to normal paragraphs.}
\DefaultSample

\StyleEight
\VariantTitle{8}{Rule-Led Blocks}{This is the largest departure: no box fill and no frame, just a title, a color rule, and a lightly indented body. It is compact but still highly scannable.}
\RuleSample
